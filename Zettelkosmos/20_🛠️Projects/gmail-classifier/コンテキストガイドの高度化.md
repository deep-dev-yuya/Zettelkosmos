
"""
Advanced Context Enricher - PDFの仕様に準拠した拡張版
エンティティ抽出、優先度判定、期限抽出を高度化
"""

from flask import Blueprint, request, jsonify
import re
import datetime
from typing import Dict, List, Optional, Tuple
import logging

context_enricher_bp = Blueprint('context_enricher', __name__)

class AdvancedContextEnricher:
    """高度な文脈補完クラス"""
    
    def __init__(self):
        # 金額抽出パターン
        self.amount_patterns = [
            r'(\d{1,3}(?:,\d{3})*|\d+)円',
            r'(\d{1,3}(?:,\d{3})*|\d+)¥',
            r'¥(\d{1,3}(?:,\d{3})*|\d+)',
            r'金額[：:]\s*(\d{1,3}(?:,\d{3})*|\d+)円?'
        ]
        
        # 日付抽出パターン（拡張版）
        self.date_patterns = [
            r'(\d{4})[年/-](\d{1,2})[月/-](\d{1,2})日?',
            r'(\d{1,2})[月/-](\d{1,2})日?',
            r'(\d{1,2})/(\d{1,2})/(\d{4})',
            r'(\d{1,2})-(\d{1,2})-(\d{4})',
            r'期限[：:]\s*(\d{4})[年/-](\d{1,2})[月/-](\d{1,2})日?'
        ]
        
        # 組織・サービス名
        self.services = [
            'PayPay', 'LINE Pay', 'Amazon', 'Google', 'Apple', 'Netflix',
            'Spotify', 'クレジットカード', 'セブンイレブン', 'ローソン',
            'ファミマ', 'イオン', 'TSUTAYA', 'ヤマト運輸', '佐川急便'
        ]
        
        # 緊急度キーワード
        self.urgency_keywords = {
            'high': ['緊急', '至急', '重要', 'urgent', 'emergency', '障害', '停止'],
            'medium': ['お知らせ', 'notification', '通知', '案内', 'ご連絡'],
            'low': ['プロモーション', 'キャンペーン', 'セール', 'お得']
        }
        
        # 支払い関連キーワード
        self.payment_keywords = [
            '支払い', '請求', '料金', '引き落とし', '決済', '入金',
            'チャージ', '残高', '利用完了', '購入', '課金'
        ]

    def extract_entities(self, text: str) -> Dict:
        """エンティティ抽出"""
        entities = {
            'amounts': [],
            'dates': [],
            'services': [],
            'urgency_level': 'low'
        }
        
        # 金額抽出
        for pattern in self.amount_patterns:
            matches = re.findall(pattern, text)
            entities['amounts'].extend(matches)
        
        # 日付抽出
        for pattern in self.date_patterns:
            matches = re.findall(pattern, text)
            entities['dates'].extend(matches)
        
        # サービス名抽出
        for service in self.services:
            if service in text:
                entities['services'].append(service)
        
        # 緊急度判定
        for level, keywords in self.urgency_keywords.items():
            if any(keyword in text for keyword in keywords):
                entities['urgency_level'] = level
                break
        
        return entities

    def analyze_payment_context(self, text: str) -> Dict:
        """支払い関連文脈の詳細分析"""
        payment_context = {
            'is_payment_related': False,
            'payment_type': None,
            'action_required': False,
            'keywords_found': []
        }
        
        # 支払い関連キーワード検索
        found_keywords = [kw for kw in self.payment_keywords if kw in text]
        payment_context['keywords_found'] = found_keywords
        
        if found_keywords:
            payment_context['is_payment_related'] = True
            
            # 支払いタイプ判定
            if any(kw in text for kw in ['チャージ', '入金']):
                payment_context['payment_type'] = 'charge'
            elif any(kw in text for kw in ['引き落とし', '決済', '利用完了']):
                payment_context['payment_type'] = 'debit'
            elif any(kw in text for kw in ['請求', '料金']):
                payment_context['payment_type'] = 'billing'
            
            # アクション要求判定
            if any(kw in text for kw in ['期限', '確認', '手続き']):
                payment_context['action_required'] = True
        
        return payment_context

    def calculate_priority(self, entities: Dict, payment_context: Dict) -> str:
        """優先度計算の高度化"""
        score = 0
        
        # 緊急度による加点
        urgency_scores = {'high': 30, 'medium': 10, 'low': 0}
        score += urgency_scores.get(entities['urgency_level'], 0)
        
        # 支払い関連による加点
        if payment_context['is_payment_related']:
            score += 20
            if payment_context['action_required']:
                score += 10
        
        # 金額による加点
        if entities['amounts']:
            score += 15
        
        # 日付による加点
        if entities['dates']:
            score += 10
        
        # 優先度判定
        if score >= 30:
            return 'high'
        elif score >= 15:
            return 'medium'
        else:
            return 'low'

    def enrich_context(self, subject: str, body: str) -> Dict:
        """統合文脈補完"""
        full_text = f"{subject} {body}"
        
        # エンティティ抽出
        entities = self.extract_entities(full_text)
        
        # 支払い文脈分析
        payment_context = self.analyze_payment_context(full_text)
        
        # 優先度計算
        priority = self.calculate_priority(entities, payment_context)
        
        # 文脈要約生成
        context_summary = self._generate_context_summary(
            entities, payment_context, priority
        )
        
        return {
            'enriched_context': context_summary,
            'priority_level': priority,
            'entities': entities,
            'payment_analysis': payment_context,
            'keywords': payment_context['keywords_found'],
            'deadline_info': entities['dates'][0] if entities['dates'] else None,
            'amount_info': entities['amounts'][0] if entities['amounts'] else None
        }

    def _generate_context_summary(self, entities: Dict, payment_context: Dict, priority: str) -> str:
        """文脈要約の生成"""
        summary_parts = []
        
        # 優先度情報
        priority_labels = {'high': '高優先度', 'medium': '中優先度', 'low': '低優先度'}
        summary_parts.append(f"{priority_labels[priority]}のメール")
        
        # 支払い関連情報
        if payment_context['is_payment_related']:
            if payment_context['payment_type'] == 'charge':
                summary_parts.append("チャージ・入金に関する通知")
            elif payment_context['payment_type'] == 'debit':
                summary_parts.append("決済・利用完了の通知")
            elif payment_context['payment_type'] == 'billing':
                summary_parts.append("請求・料金に関する通知")
            else:
                summary_parts.append("支払い関連の通知")
        
        # 金額情報
        if entities['amounts']:
            summary_parts.append(f"金額: {entities['amounts'][0]}円")
        
        # サービス情報
        if entities['services']:
            summary_parts.append(f"サービス: {', '.join(entities['services'])}")
        
        # 緊急度情報
        if entities['urgency_level'] == 'high':
            summary_parts.append("緊急対応が必要")
        
        return "、".join(summary_parts)

# グローバルインスタンス
enricher = AdvancedContextEnricher()

@context_enricher_bp.route('/enrich-context', methods=['POST'])
def enrich_context():
    """文脈補完APIエンドポイント"""
    try:
        data = request.json
        
        # 入力検証
        if not data or 'subject' not in data or 'body' not in data:
            return jsonify({
                "error": "Missing required fields: subject, body"
            }), 400
        
        subject = data.get('subject', '')
        body = data.get('body', '')
        
        # 文脈補完実行
        result = enricher.enrich_context(subject, body)
        
        return jsonify(result)
        
    except Exception as e:
        logging.error(f"Context enrichment failed: {str(e)}")
        return jsonify({
            "error": f"Context enrichment failed: {str(e)}"
        }), 500

@context_enricher_bp.route('/context/analyze-payment', methods=['POST'])
def analyze_payment():
    """支払い分析専用エンドポイント"""
    try:
        data = request.json
        text = f"{data.get('subject', '')} {data.get('body', '')}"
        
        payment_analysis = enricher.analyze_payment_context(text)
        entities = enricher.extract_entities(text)
        
        return jsonify({
            'payment_analysis': payment_analysis,
            'extracted_amounts': entities['amounts'],
            'extracted_dates': entities['dates'],
            'services_mentioned': entities['services']
        })
        
    except Exception as e:
        return jsonify({
            "error": f"Payment analysis failed: {str(e)}"
        }), 500
